#### Fix autoinclude inputs overriding unit inputs that reference undefined values

Fixes an issue where unit inputs referencing undefined `values.*` attributes failed HCL decoding before autoinclude inputs could override them. When a sibling autoinclude is present, decoding now defers those attribute errors to the autoinclude merge, which overrides the affected inputs as expected.

```hcl
# stacks/terragrunt.stack.hcl
unit "subnet" {
  source = "../units/subnet"
  path   = "subnet"

  autoinclude {
    dependency "vpc" {
      config_path  = unit.vpc.path
      mock_outputs = { vpc_id = "mock" }
    }

    inputs = {
      vpc_id = dependency.vpc.outputs.vpc_id
    }
  }

  values = {
    cidr_block = "10.0.0.0/24"
  }
}
```

```hcl
# units/subnet/terragrunt.hcl
inputs = {
  vpc_id     = values.vpc_id     # decode deferred, overridden by autoinclude
  cidr_block = values.cidr_block # still resolves from the values file
}
```
